fdt_high=ffffffffffffffff
initrd_high=ffffffffffffffff

# Bare bsp-linux on Verdin iMX8MP: SPL → BL31 → U-Boot (no OP-TEE, no
# AVZ). U-Boot bootm's a single ITB carrying kernel + DTB + buildroot
# initrd. TEZI installs the ITB on the otaroot partition (mmc 2:1).

LOAD_ADDR=0x90000000

start=ext4load mmc 2:1 ${LOAD_ADDR} verdin-imx8mp.itb; bootm ${LOAD_ADDR}

bootargs=earlycon=ec_imx6q,0x30880000,115200 console=ttymxc2,115200 root=/dev/ram coherent_pool=1M cma=256M rootwait vt.global_cursor_default=0

fdt_high=ffffffffffffffff
initrd_high=ffffffffffffffff

# Bare bsp-torizon on virt64: ATF + U-Boot (no OP-TEE, no AVZ) in
# flash0.img. U-Boot bootm's a single ITB carrying our kernel + DTB +
# the TorizonOS OSTree-aware initramfs. The initramfs reads ostree=...
# and pivots to the deployment on p2 as the real root.

LOAD_ADDR=0x65000000

start=load virtio 0:1 ${LOAD_ADDR} virt64.itb; bootm ${LOAD_ADDR}

bootargs=earlycon=pl011,mmio32,0x09000000 console=ttyAMA0,115200n8 root=/dev/vda2 rootfstype=ext4 ostree=/ostree/boot.1/torizon/${OSTREE_DEPLOYMENT_ID}/0

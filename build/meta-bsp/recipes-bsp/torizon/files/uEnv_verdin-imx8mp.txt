fdt_high=ffffffffffffffff
initrd_high=ffffffffffffffff

# Bare bsp-torizon on Verdin iMX8MP: SPL → BL31 → U-Boot (no OP-TEE,
# no AVZ). U-Boot bootm's a single ITB carrying our kernel + DTB +
# the TorizonOS OSTree-aware initramfs. The initramfs reads ostree=...
# and pivots to the deployment on the otaroot partition (mmc 2:1).

LOAD_ADDR=0x90000000

start=ext4load mmc 2:1 ${LOAD_ADDR} verdin-imx8mp.itb; bootm ${LOAD_ADDR}

bootargs=earlycon=ec_imx6q,0x30880000,115200 console=ttymxc2,115200 root=/dev/mmcblk2p1 rootfstype=ext4 ostree=/ostree/boot.1/torizon/${OSTREE_DEPLOYMENT_ID}/0 systemd.show_status=true
